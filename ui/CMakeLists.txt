add_library(penguin_ui STATIC src/core.c)
target_include_directories(penguin_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(penguin_ui PUBLIC c_std_17)
if(NOT MSVC)
    target_compile_options(penguin_ui PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(penguin_ui PUBLIC m)
endif()
function(penguin_ui_add_sdl)
    if(NOT TARGET penguin_ui_sdl)
        add_library(penguin_ui_sdl STATIC ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/src/sdl.c)
        target_link_libraries(penguin_ui_sdl PUBLIC penguin_ui SDL3::SDL3)
        if(NOT MSVC)
            target_compile_options(penguin_ui_sdl PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endfunction()
if(TARGET SDL3::SDL3)
    penguin_ui_add_sdl()
endif()
